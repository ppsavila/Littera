**B. Create WelcomeModalTrigger client component:**

Create `src/components/subscription/WelcomeModalTrigger.tsx`:
```typescript
'use client'

import { useState, useEffect } from 'react'
import { UpgradeModal } from './UpgradeModal'

interface Props {
  onboarded: boolean
  currentPlan?: 'free' | 'plus' | 'premium'
}

export function WelcomeModalTrigger({ onboarded, currentPlan = 'free' }: Props) {
  const [open, setOpen] = useState(false)

  useEffect(() => {
    if (!onboarded) {
      setOpen(true)
    }
  }, [onboarded])

  async function handleClose() {
    setOpen(false)
    // Mark as onboarded — fire and forget, optimistic
    try {
      await fetch('/api/profile', {
        method: 'PATCH',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ onboarded: true }),
      })
    } catch {
      // Silent fail — modal won't reappear due to local state
    }
  }

  return <UpgradeModal open={open} onClose={handleClose} reason="welcome" currentPlan={currentPlan} />
}
```

Key design points:
- `useEffect` opens modal on mount when `onboarded === false` (not on every render)
- PATCH fires only in `handleClose` (on dismiss), not on mount — avoids Pitfall 5
- Local `open` state set to `false` immediately on close — modal won't flash back even if parent re-renders
